# bash completion for kanban-cli
_kanban_cli_completions() {
    local cur prev words cword
    _init_completion || return

    local top_commands="search-tickets read-ticket create-ticket update-ticket begin-work transition-ticket complete-work add-comment link-objects docs admin auth completions"
    local docs_commands="search create update"
    local admin_commands="configure-workspace manage-auth-cards manage-principals-and-groups manage-access-policies architect-project manage-cycles-and-modules manage-custom-fields-and-scripts manage-workflow-interceptors manage-ai-and-agent-runners override-ticket-provenance audit-and-purge-object-links destructive-lifecycle"
    local auth_commands="login status logout"

    if [[ $cword -eq 1 ]]; then
        COMPREPLY=( $(compgen -W "$top_commands --url --workspace --project --token --json --jsonl --help" -- "$cur") )
        return 0
    fi

    case "${words[1]}" in
        docs)
            COMPREPLY=( $(compgen -W "$docs_commands" -- "$cur") )
            return 0
            ;;
        admin)
            COMPREPLY=( $(compgen -W "$admin_commands" -- "$cur") )
            return 0
            ;;
        auth)
